# Script to generate 1000 random regions and calculate coverage using Mosdepth.  
# Change the `-n` value in bedtools command to match expected region counts for different species.  

#!/bin/bash
# Step 1: Generate 1000 random seed numbers
shuf -i 1-1000000 -n 1000 > random_seeds.txt

# Step 2: Prepare the genome file
cp ../nuclear_genome.fasta .
samtools faidx nuclear_genome.fasta
cut -f1,2 nuclear_genome.fasta.fai > nuclear.genome

# Step 3: Generate random region files and run mosdepth
seeds=$(cat random_seeds.txt)

echo "Activating mosdepth environment..."
source activate mosdepth  # or use: conda activate mosdepth

for seed in $seeds
do
echo "Generating region_${seed}.bed ..."
/home/cdb3ny/anaconda3/bin/bedtools random -l 250 -n 722  -g nuclear.genome -seed "$seed" > "region_${seed}.bed" #change -n value for different species
echo "Running mosdepth for region_${seed}.bed ..."
mosdepth -n --by "region_${seed}.bed" --threads 11 "mosdepth_region_${seed}" ../Minimap2/chromosome_alignment_sorted.bam
done

echo "Deactivating mosdepth..."
conda deactivate

echo "All tasks completed successfully!"
